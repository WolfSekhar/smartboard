  <script>
    // --- 1. HOST DEVICE TELEMETRY BRIDGE ---
    let hostDpr = window.devicePixelRatio || 1;
    window.addEventListener('message', (e) => {
      if (!e.data || typeof e.data !== 'object') return;
      if (e.data.type === 'HOST_FRAME_DATA') {
        if (e.data.device && e.data.device.dpr) { hostDpr = e.data.device.dpr; resizeCanvas(); }
        if (e.data.theme) {
          isLightTheme = (e.data.theme === 'light');
          document.body.classList.toggle('light-theme', isLightTheme);
          updateThemeIcon();
          updateAllSliderFills();
        }
      }
    });

    // --- 2. WEBGPU ACCELERATION PIPELINE (PROGRESSIVE ENHANCEMENT WITH FALLBACK) ---
    let isWebGPUSupported = false;
    let gpuDevice = null;

    async function initWebGPU() {
      if (!navigator.gpu) return;
      try {
        const adapter = await navigator.gpu.requestAdapter({ powerPreference: 'high-performance' });
        if (!adapter) return;
        gpuDevice = await adapter.requestDevice();
        if (!gpuDevice) return;
        isWebGPUSupported = true;
      } catch (err) {
        console.info('WebGPU optional initialization bypassed, using Canvas 2D engine.', err);
      }
    }
    initWebGPU();

    // --- 3. UI STATE & 1000 HZ VSYNC MICRO-STEPPING ENGINE ---
    const canvas = document.getElementById('simCanvas');
    const ctx = canvas.getContext('2d', { alpha: false });
    const container = document.getElementById('canvasStageContainer');
    let canvasWidth = 500; let canvasHeight = 320;
    let isLightTheme = true; let isRunning = false; // Starts paused on initial load (Adwaita Light Mode Default)
    let zoomScale = 1.0; let panOffset = { x: 0, y: 0 };
    const toolStates = { cornerGraph: false, ruler: false };
    const MAX_POINTS = 200;
    let waveHistory = [];

    // Physics parameters & state
    let p1 = 1.0;
    let speedMultiplier = 1.0;
    let state = 0;
    let simTime = 0;
    let physicsAccumulator = 0;
    let lastFrameTime = performance.now();
    const MICRO_DT = 0.001; // 1 ms micro-steps (1000 Hz exact symplectic integration)

    let liveState = { time: 0, state: 0 };

    // --- ZERO-OVERHEAD SLIDER ACTIVE TRACK HIGHLIGHT (O(1)) ---
    function updateSliderFill(slider) {
      if (!slider) return;
      const min = parseFloat(slider.min || 0);
      const max = parseFloat(slider.max || 100);
      const val = parseFloat(slider.value);
      const pct = Math.max(0, Math.min(100, ((val - min) / (max - min)) * 100));
      slider.style.background = `linear-gradient(to right, var(--adw-scale-track-fill) 0%, var(--adw-scale-track-fill) ${pct}%, var(--adw-scale-track-bg) ${pct}%, var(--adw-scale-track-bg) 100%)`;
    }

    function updateAllSliderFills() {
      document.querySelectorAll('.adw-scale-slider').forEach(updateSliderFill);
    }

    function updateThemeIcon() {
      const btn = document.getElementById('themeToggleBtn');
      if (!btn) return;
      btn.innerHTML = isLightTheme 
        ? `<svg class="lucide-icon" width="13" height="13" viewBox="0 0 24 24"><path d="M12 3a6 6 0 0 0 9 9 9 9 0 1 1-9-9Z"/></svg>`
        : `<svg class="lucide-icon" width="13" height="13" viewBox="0 0 24 24"><circle cx="12" cy="12" r="4"/><path d="M12 2v2"/><path d="M12 20v2"/><path d="m4.93 4.93 1.41 1.41"/><path d="m17.66 17.66 1.41 1.41"/><path d="M2 12h2"/><path d="M20 12h2"/><path d="m6.34 17.66-1.41 1.41"/><path d="m19.07 4.93-1.41 1.41"/></svg>`;
    }

    function toggleTheme() {
      isLightTheme = !isLightTheme;
      document.body.classList.toggle('light-theme', isLightTheme);
      updateThemeIcon();
      updateAllSliderFills();
    }

    function togglePlay() {
      isRunning = !isRunning;
      const btn = document.getElementById('playBtn');
      if (btn) {
        btn.innerHTML = isRunning
          ? `<svg class="lucide-icon" width="12" height="12" viewBox="0 0 24 24"><rect x="6" y="4" width="4" height="16" rx="1"/><rect x="14" y="4" width="4" height="16" rx="1"/></svg><span>Pause</span>`
          : `<svg class="lucide-icon" width="12" height="12" viewBox="0 0 24 24" fill="currentColor" stroke="none"><polygon points="6 3 20 12 6 21 6 3"/></svg><span>Play</span>`;
      }
      lastFrameTime = performance.now();
    }

    function restartSim() {
      state = 0;
      simTime = 0;
      physicsAccumulator = 0;
      lastFrameTime = performance.now();
      waveHistory = [];
    }

    function resetToDefaults() {
      document.getElementById('p1Slider').value = 1.0;
      document.getElementById('speedSlider').value = 1.0;
      p1 = 1.0;
      speedMultiplier = 1.0;
      updateParams();
      restartSim();
    }

    function updateParams() {
      p1 = parseFloat(document.getElementById('p1Slider').value);
      document.getElementById('p1Val').innerText = p1.toFixed(2);
      updateAllSliderFills();
    }

    function setSpeed(val) {
      speedMultiplier = parseFloat(val);
      document.getElementById('speedVal').innerText = speedMultiplier.toFixed(2) + 'x';
      updateSliderFill(document.getElementById('speedSlider'));
    }

    function toggleTool(name) {
      toolStates[name] = !toolStates[name];
      const card = document.getElementById('cornerGraphWidget');
      const pill = document.getElementById('pillCornerGraph');
      if (card) card.classList.toggle('open', toolStates[name]);
      if (pill) pill.classList.toggle('active', toolStates[name]);
    }

    function toggleRuler() {
      toolStates.ruler = !toolStates.ruler;
      document.getElementById('pillRuler')?.classList.toggle('active', toolStates.ruler);
    }

    function resetCanvasZoom() {
      zoomScale = 1.0; panOffset = { x: 0, y: 0 };
      document.getElementById('zoomPercentText').innerText = '100%';
    }

    function resizeCanvas() {
      const rect = container.getBoundingClientRect();
      if (!rect.width || !rect.height) return;
      canvasWidth = rect.width; canvasHeight = rect.height;
      const dpr = hostDpr || window.devicePixelRatio || 1;
      canvas.width = Math.round(canvasWidth * dpr);
      canvas.height = Math.round(canvasHeight * dpr);
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    }

    if (window.ResizeObserver) {
      const ro = new ResizeObserver(() => resizeCanvas());
      ro.observe(container);
    } else {
      window.addEventListener('resize', resizeCanvas);
    }

    // --- 4. 1000 HZ VSYNC MICRO-STEPPING INTEGRATION & HIGH-FPS RENDER LOOP ---
    function draw() {
      if (canvasWidth <= 0 || canvas.width === 0) {
        resizeCanvas();
      }

      const now = performance.now();
      let dt_real = (now - lastFrameTime) / 1000;
      lastFrameTime = now;
      if (dt_real > 0.1) dt_real = 0.1; // Safety clamp against background tab pause

      // 1000 Hz VSync Micro-Stepping
      if (isRunning) {
        physicsAccumulator += dt_real * speedMultiplier;
        while (physicsAccumulator >= MICRO_DT) {
          // [TOPIC-SPECIFIC PHYSICS DIFFERENTIAL EQUATION INTEGRATION]
          state = Math.sin(simTime * p1 * 2.0);
          simTime += MICRO_DT;
          physicsAccumulator -= MICRO_DT;
        }

        liveState = { time: simTime, state };
        waveHistory.push({ t: simTime, val: state });
        if (waveHistory.length > MAX_POINTS) waveHistory.shift();
      }

      ctx.fillStyle = isLightTheme ? '#ffffff' : '#242424';
      ctx.fillRect(0, 0, canvasWidth, canvasHeight);

      // MANDATORY SUBTLE CARTESIAN GRID (STATIC SCREEN SPACE, SINGLE BATCHED PATH)
      ctx.strokeStyle = isLightTheme ? '#f0f0f0' : '#333333';
      ctx.lineWidth = 1;
      ctx.beginPath();
      for (let x = 0; x < canvasWidth; x += 40) { ctx.moveTo(x, 0); ctx.lineTo(x, canvasHeight); }
      for (let y = 0; y < canvasHeight; y += 40) { ctx.moveTo(0, y); ctx.lineTo(canvasWidth, y); }
      ctx.stroke();

      // Physical Apparatus Rendering (Layer 2: Zoom & Pan Matrix)
      ctx.save();
      ctx.translate(canvasWidth / 2 + panOffset.x, canvasHeight / 2 + panOffset.y);
      ctx.scale(zoomScale, zoomScale);

      ctx.fillStyle = isLightTheme ? '#1c71d8' : '#3584e4';
      ctx.beginPath();
      ctx.arc(liveState.state * 80, 0, 16, 0, Math.PI * 2);
      ctx.fill();

      ctx.restore();
      requestAnimationFrame(draw);
    }

    // --- RESIZABLE SIDEBAR SPLITTER LOGIC ---
    const splitter = document.getElementById('paneSplitter');
    const controlsPanel = document.querySelector('.controls-panel');
    let isResizing = false;

    if (splitter && controlsPanel) {
      splitter.addEventListener('pointerdown', (e) => {
        isResizing = true;
        splitter.classList.add('dragging');
        splitter.setPointerCapture(e.pointerId);
        document.body.style.cursor = 'col-resize';
        document.body.style.userSelect = 'none';
      });

      splitter.addEventListener('pointermove', (e) => {
        if (!isResizing) return;
        const containerRect = document.querySelector('.sim-layout').getBoundingClientRect();
        const newWidth = Math.max(200, Math.min(500, e.clientX - containerRect.left - 3));
        controlsPanel.style.width = newWidth + 'px';
        controlsPanel.style.minWidth = newWidth + 'px';
        controlsPanel.style.maxWidth = newWidth + 'px';
        controlsPanel.style.flex = `0 0 ${newWidth}px`;
      });

      const stopResize = (e) => {
        if (!isResizing) return;
        isResizing = false;
        splitter.classList.remove('dragging');
        try { splitter.releasePointerCapture(e.pointerId); } catch(err) {}
        document.body.style.cursor = '';
        document.body.style.userSelect = '';
      };

      splitter.addEventListener('pointerup', stopResize);
      splitter.addEventListener('pointercancel', stopResize);

      splitter.addEventListener('dblclick', () => {
        controlsPanel.style.width = '270px';
        controlsPanel.style.minWidth = '270px';
        controlsPanel.style.maxWidth = '270px';
        controlsPanel.style.flex = '0 0 270px';
      });
    }

    window.addEventListener('DOMContentLoaded', () => {
      resizeCanvas();
      document.querySelectorAll('.adw-scale-slider').forEach(slider => {
        slider.addEventListener('input', () => updateSliderFill(slider));
      });
      updateParams();
      updateThemeIcon();
      lastFrameTime = performance.now();
      requestAnimationFrame(draw);
    });
  </script>
